   35: =head1 SYNOPSIS
   36: 
   37: Hold many texts in one object,
   38: thereby encapsulating internationalized text.
   39: 
   40:     use Lingua::Text;
   41: 
   42:     my $str = Lingua::Text->new();
   43: 
   44:     $str->fr('Bonjour Tout le Monde');
   45:     $str->en('Hello, World');
   46: 
   47:     $ENV{'LANG'} = 'en_GB';
   48:     print "$str\n";	# Prints Hello, World
   49:     $ENV{'LANG'} = 'fr_FR';
   50:     print "$str\n";	# Prints Bonjour Tout le Monde
   51:     $ENV{'LANG'} = 'de_DE';
   52:     print "$str\n";	# Prints nothing
   53: 
   54:     my $text = Lingua::Text->new('hello');	# Initialises the 'current' language
